# === 공개 데모(Streamlit Cloud) 의존성 ===
#
# Streamlit Cloud가 저장소 루트의 이 파일을 자동으로 설치하므로,
# 데모에 필요한 최소 구성만 담는다.
#
# ▶ 온프레미스(사내 폐쇄망) 설치는 `requirements-onprem.txt`를 사용한다.
#     pip install -r requirements-onprem.txt
#
# 데모에 torch·sentence-transformers·fastapi·ollama를 넣지 않는 이유:
#   - torch/sentence-transformers(bge-m3)는 Cloud 메모리 한도를 초과한다.
#   - fastapi/uvicorn은 Streamlit과 starlette 버전이 충돌한다.
#   - 로컬 LLM(Ollama)은 Cloud에서 구동할 수 없다.
# 라이브러리 버전은 온프레미스와 동일하게 맞춰 이식 시 동작 차이를 줄인다.

streamlit==1.40.1

# LangChain (온프레미스와 동일 계열)
langchain-openai==0.2.9
langchain-qdrant==0.2.0
langchain-text-splitters==0.3.2
# langsmith(추적 기능, 이 프로젝트는 사용하지 않음)는 langchain-core의 전이 의존성이다.
# 0.5+ 는 uuid-utils, 0.6.7+ 는 xxhash 같은 네이티브 모듈을 추가로 끌어오는데,
# Windows 애플리케이션 제어 정책이 이런 DLL을 차단하면 임포트 단계에서 죽는다.
# 네이티브 의존성이 가장 적은 0.4.x 로 고정한다. (폐쇄망 반입 wheel도 줄어든다)
langsmith==0.4.31

# 벡터DB (Docker 없이 메모리/파일 모드로 사용)
qdrant-client==1.12.1

# 경량 로컬 임베딩 (ONNX, torch 불필요)
